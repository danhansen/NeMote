cmake_minimum_required(VERSION 3.20)
project(nemote_service LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
set(NEMO_SPEECH_SOURCE_DIR "" CACHE PATH "Pinned NeMo checkout (JSON header)")
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp PATHS "${NEMO_SPEECH_SOURCE_DIR}/llama.cpp/vendor" REQUIRED NO_DEFAULT_PATH)
# The extension ships this same protocol file; don't maintain a second copy.
set(SERVICE_PROTOCOL "${CMAKE_CURRENT_SOURCE_DIR}/../../extensions/gnome-shell/nemote@dhansen.dev/service.xml")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SERVICE_PROTOCOL}")
file(READ "${SERVICE_PROTOCOL}" SERVICE_XML)
configure_file(interface.h.in interface.h @ONLY)
add_executable(nemote-service main.cpp)
target_include_directories(nemote-service PRIVATE "${JSON_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(nemote-service PRIVATE PkgConfig::GIO)
target_compile_options(nemote-service PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS nemote-service RUNTIME DESTINATION bin)
enable_testing()
add_executable(test-service-core test_core.cpp)
target_include_directories(test-service-core PRIVATE "${JSON_INCLUDE_DIR}")
add_test(NAME service-core COMMAND test-service-core)
