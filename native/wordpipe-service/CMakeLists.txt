cmake_minimum_required(VERSION 3.20)
project(wordpipe_service LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GIO REQUIRED IMPORTED_TARGET gio-unix-2.0)
set(NEMO_SPEECH_SOURCE_DIR "" CACHE PATH "Pinned NeMo checkout (JSON header)")
find_path(JSON_INCLUDE_DIR nlohmann/json.hpp PATHS "${NEMO_SPEECH_SOURCE_DIR}/llama.cpp/vendor" REQUIRED NO_DEFAULT_PATH)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/interface.xml" SERVICE_XML)
configure_file(interface.h.in interface.h @ONLY)
add_executable(wordpipe-service main.cpp)
target_include_directories(wordpipe-service PRIVATE "${JSON_INCLUDE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(wordpipe-service PRIVATE PkgConfig::GIO)
target_compile_options(wordpipe-service PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS wordpipe-service RUNTIME DESTINATION bin)
enable_testing()
add_executable(test-service-core test_core.cpp)
target_include_directories(test-service-core PRIVATE "${JSON_INCLUDE_DIR}")
add_test(NAME service-core COMMAND test-service-core)
